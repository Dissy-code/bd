Процедура ПриСозданииНаСервере(Отказ, СтандартнаяОбработка)
	// Подставляем значения по умолчанию при открытии формы
	Если Не ЗначениеЗаполнено(Объект.Сервер) Тогда

	КонецЕсли;
	Если Объект.Порт = 0 Тогда
		Объект.Порт = 8000; // порт, на котором слушает uvicorn
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Объект.Путь) Тогда
		Объект.Путь = "/notes"; // адрес эндпоинта
	КонецЕсли;
КонецПроцедуры

&НаСервере
Процедура СинхронизироватьНаСервере()
	Т0 = ТекущаяДата(); // отметка времени: старт

	Защищенное = (Объект.Порт = 443); // HTTPS только для порта 443
	Соединение = Новый HTTPСоединение(Объект.Сервер, Объект.Порт,,,,, Защищенное); // открываем соединение
	Т1 = ТекущаяДата(); // отметка: соединение создано

	Ответ = Соединение.Получить(Новый HTTPЗапрос(Объект.Путь)); // сам GET-запрос
	Т2 = ТекущаяДата(); // отметка: ответ получен

	МассивЗаметок = ПрочитатьЗначениеJSON(Ответ.ПолучитьТелоКакСтроку()); // текст -> массив структур
	Т3 = ТекущаяДата(); // отметка: JSON разобран

	Для Каждого Заметка Из МассивЗаметок Цикл // перебор заметок по одной
		Код = Заметка.id; // код заметки из JSON
		ЗаголовокПользователь = Заметка.title_user; // заголовок + логин
		Содержание = Заметка.content; // текст заметки

		Ссылка = Справочники.Заметки.НайтиПоКоду(Код); // ищем по коду
		Если Ссылка.Пустая() Тогда
			Элемент = Справочники.Заметки.СоздатьЭлемент(); // не нашли — создаём
			Элемент.Код = Код;
		Иначе
			Элемент = Ссылка.ПолучитьОбъект(); // нашли — берём на редактирование
		КонецЕсли;

		Элемент.Наименование = ЗаголовокПользователь;
		Элемент.Содержание = Содержание;
		Элемент.Записать(); // сохраняем в базу
	КонецЦикла;

	Т4 = ТекущаяДата(); // отметка: запись завершена

	// Итоговое сообщение с разбивкой по времени (для диагностики скорости)
	Сообщить("Синхронизация завершена. Создание соединения: "
		+ (Т1 - Т0) + " сек, HTTP-запрос: " + (Т2 - Т1)
		+ " сек, Разбор JSON: " + (Т3 - Т2) + " сек, Запись в БД: " + (Т4 - Т3) + " сек");
КонецПроцедуры
